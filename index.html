// ============================================================ // ⏱️ AUTOMATIC SCHEDULE SYSTEM // ============================================================ const SCHEDULE_DAY_LABELS = [ 'الأحد', 'الاثنين', 'الثلاثاء', 'الأربعاء', 'الخميس', 'الجمعة', 'السبت' ]; const DEFAULT_SCHEDULE = { enabled: false, days: [0,1,2,3,4,5,6], start: '00:00', end: '23:59', countdown: true, oneDate: '' }; function normalizeSchedule(schedule){ if(!schedule || typeof schedule !== 'object'){ return {...DEFAULT_SCHEDULE}; } const days = Array.isArray(schedule.days) ? schedule.days .map(Number) .filter(n => Number.isInteger(n) && n >= 0 && n <= 6 ) : DEFAULT_SCHEDULE.days; return { enabled: schedule.enabled === true, days: [...new Set(days)], start: /^\d{2}:\d{2}$/.test(schedule.start || '') ? schedule.start : '00:00', end: /^\d{2}:\d{2}$/.test(schedule.end || '') ? schedule.end : '23:59', countdown: schedule.countdown !== false, oneDate: schedule.oneDate || '' }; } function ensureScheduleDefaults(){ MENU.forEach(cat => { cat.schedule = normalizeSchedule(cat.schedule); cat.items = Array.isArray(cat.items) ? cat.items : []; cat.items.forEach(item => { item.schedule = normalizeSchedule(item.schedule); }); }); } function timeToMinutes(value){ const match = String(value || '') .match(/^(\d{2}):(\d{2})$/); if(!match) return 0; return Number(match[1]) * 60 + Number(match[2]); } function isScheduleActive(schedule, now = new Date()){ const s = normalizeSchedule(schedule); // الجدولة غير مفعلة = العنصر شغال عادي if(!s.enabled){ return true; } // لو فيه تاريخ محدد if(s.oneDate){ const ymd = now.getFullYear() + '-' + String(now.getMonth()+1).padStart(2,'0') + '-' + String(now.getDate()).padStart(2,'0'); if(ymd !== s.oneDate){ return false; } }else{ // لو مفيش تاريخ محدد استخدم أيام الأسبوع if(!s.days.includes(now.getDay())){ return false; } } const current = now.getHours() * 60 + now.getMinutes(); const start = timeToMinutes(s.start); const end = timeToMinutes(s.end); // نفس الوقت = شغال طول اليوم if(start === end){ return true; } // الفترة عادية if(end > start){ return current >= start && current < end; } // الفترة بتعدي بعد منتصف الليل return current >= start || current < end; } function getScheduleEndDate(schedule, now = new Date()){ const s = normalizeSchedule(schedule); if(!s.enabled){ return null; } const endMin = timeToMinutes(s.end); const startMin = timeToMinutes(s.start); const end = new Date(now); end.setSeconds(0,0); end.setHours( Math.floor(endMin / 60), endMin % 60, 0, 0 ); if(end <= now){ end.setDate(end.getDate() + 1); } if( endMin < startMin && now.getHours() * 60 + now.getMinutes() < endMin ){ end.setDate(end.getDate() - 1); } if(s.oneDate){ const [y,m,d] = s.oneDate.split('-').map(Number); end.setFullYear(y); end.setMonth(m - 1); end.setDate(d); } return end; } function formatRemaining(milliseconds){ let seconds = Math.max( 0, Math.floor(milliseconds / 1000) ); const days = Math.floor(seconds / 86400); seconds %= 86400; const hours = Math.floor(seconds / 3600); seconds %= 3600; const minutes = Math.floor(seconds / 60); seconds %= 60; if(days){ return `${days} يوم ${hours} س`; } if(hours){ return `${hours} س ${String(minutes).padStart(2,'0')} د`; } return `${String(minutes).padStart(2,'0')}:${String(seconds).padStart(2,'0')}`; } function scheduleCountdownHtml( schedule, now = new Date() ){ const s = normalizeSchedule(schedule); if( !s.enabled || !s.countdown || !isScheduleActive(s, now) ){ return ''; } const end = getScheduleEndDate(s, now); if(!end){ return ''; } return `
⏳ متبقي ${formatRemaining( end.getTime() - now.getTime() )}
`; } function isCategoryVisible( category, now = new Date() ){ return !!category && category.available !== false && isScheduleActive( category.schedule, now ); } function isItemVisible( item, now = new Date() ){ return !!item && item.available !== false && isScheduleActive( item.schedule, now ); } function getScheduleTarget( type, catIdx, itemIdx ){ const cat = MENU[catIdx]; if(!cat){ return null; } if(type === 'item'){ return cat.items[itemIdx]; } return cat; } function buildScheduleEditor( schedule, type, catIdx, itemIdx = '' ){ const s = normalizeSchedule(schedule); return `
⏱️ إعداد الجدولة
لو اخترت تاريخًا محددًا، الجدولة تعمل في اليوم ده. لو التاريخ فاضي، الأيام المختارة هي اللي تتحكم.
${ SCHEDULE_DAY_LABELS .map((label, day) => ` `) .join('') }
`; } function saveScheduleState(){ ensureScheduleDefaults(); localStorage.setItem( MENU_STORAGE_KEY, JSON.stringify(MENU) ); apiSet( 'menu', MENU ); } function refreshAfterScheduleChange(){ renderMenu(); renderTicket(); if( typeof buildAdminPanel === 'function' && adminPanel && adminPanel.classList.contains('open') ){ buildAdminPanel(); } if( typeof buildStaffPanel === 'function' && staffPanel && staffPanel.classList.contains('open') ){ buildStaffPanel(); } } function tickScheduleCountdowns(){ const now = new Date(); let rerender = false; document .querySelectorAll('[data-schedule-end]') .forEach(el => { const remaining = Number(el.dataset.scheduleEnd) - now.getTime(); if(remaining <= 0){ rerender = true; }else{ el.textContent = formatRemaining( remaining ); } }); if(rerender){ refreshAfterScheduleChange(); } } function handleScheduleFieldChange(e){ const el = e.target; const type = el.dataset.type; const catIdx = parseInt( el.dataset.cat, 10 ); const itemIdx = el.dataset.item === '' ? null : parseInt( el.dataset.item, 10 ); const target = getScheduleTarget( type, catIdx, itemIdx ); if(!target){ return; } const s = normalizeSchedule( target.schedule ); if( el.dataset.role === 'schedule-enabled' ){ s.enabled = el.value === 'yes'; } if( el.dataset.role === 'schedule-date' ){ s.oneDate = el.value || ''; } if( el.dataset.role === 'schedule-start' ){ s.start = el.value || '00:00'; } if( el.dataset.role === 'schedule-end' ){ s.end = el.value || '23:59'; } if( el.dataset.role === 'schedule-countdown' ){ s.countdown = el.value === 'yes'; } if( el.dataset.role === 'schedule-day' ){ const day = Number(el.value); if(el.checked){ s.days = [ ...new Set([ ...s.days, day ]) ]; }else{ s.days = s.days.filter( x => x !== day ); if(!s.days.length){ el.checked = true; s.days = [day]; } } } target.schedule = s; saveScheduleState(); refreshAfterScheduleChange(); } // يبدأ العداد كل ثانية setInterval( tickScheduleCountdowns, 1000 );
